% Proteus-1 architecture figure.
%
% Build:  tectonic -X compile docs/figures/block_diagram.tex --outdir build/figures
% The PNG used by the README is rasterised from the resulting PDF at 300 dpi
% (see the `figures` target in the top-level Makefile).
%
% Drawn as a datasheet schematic, not a poster: white sheet, square boxes with
% hairline strokes, one weight of black for every net, and short orthogonal
% stubs in the gutters between columns. Nets that would have to cross a box are
% not drawn at all; the two memories sit directly above the core port that
% addresses them, so every connection is a straight run in a gap.
%
% Solid  = data path.  Dashed = control, enable or write strobe.
\documentclass[border=10pt]{standalone}

\usepackage[T1]{fontenc}
\usepackage[scaled=0.88]{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, calc}

\definecolor{ink}{HTML}{1A1A1A}
\definecolor{body}{HTML}{3A3A3A}
\definecolor{quiet}{HTML}{77736E}
\definecolor{stroke}{HTML}{2B2B2B}
\definecolor{unitfill}{HTML}{F4F4F1}
\definecolor{corefill}{HTML}{FFFFFF}
\definecolor{padfill}{HTML}{FAFAF8}

\tikzset{
  unit/.style={
    draw=stroke, line width=0.5pt, fill=unitfill, align=center,
    inner sep=2.2pt, text=body,
    font=\sffamily\fontsize{6.3}{7.4}\selectfont,
  },
  coreunit/.style={unit, fill=corefill},
  padunit/.style={unit, fill=padfill},
  ttl/.style={font=\sffamily\bfseries\fontsize{6.7}{7.8}\selectfont, text=ink},
  sig/.style={font=\sffamily\fontsize{5.9}{6.9}\selectfont, text=quiet, inner sep=1.5pt},
  data/.style={draw=ink, line width=0.6pt, -{Stealth[length=1.6mm, width=1.1mm]}},
  both/.style={draw=ink, line width=0.6pt,
               {Stealth[length=1.6mm, width=1.1mm]}-{Stealth[length=1.6mm, width=1.1mm]}},
  ctl/.style={draw=ink, line width=0.5pt, dash pattern=on 1.6mm off 1.1mm,
              -{Stealth[length=1.6mm, width=1.1mm]}},
  flow/.style={draw=ink, line width=0.5pt, -{Stealth[length=1.2mm, width=0.9mm]}},
}

% Box title: a switch, so it can be dropped in front of a line of text.
\def\ttl{\color{ink}\bfseries}

% Bus width tick, drawn across a net.
\newcommand{\bustick}[1]{\draw[draw=ink, line width=0.5pt] ($(#1)+(-0.05,-0.09)$) -- ($(#1)+(0.05,0.09)$);}
