% Proteus-1 block diagram.
%
% Build:  tectonic -X compile docs/figures/block_diagram.tex --outdir build/figures
% The PNG used by the README is rasterised from the resulting PDF at 300 dpi.
%
% Schematic, but topologically faithful: every arrow is a net that exists in
% src/tt_um_proteus.v and it starts and ends on the module port that drives or
% samples it. `proteus_core` is a container because the PC, timer and capture
% control are registers inside it rather than separate modules.
%
% Solid  = data path.
% Dashed = control / enables / write strobes.
\documentclass[border=0pt]{standalone}

\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, calc}

\definecolor{bg}{HTML}{08111F}
\definecolor{boxfill}{HTML}{0E1C2F}
\definecolor{boxedge}{HTML}{6EA8D4}
\definecolor{ink}{HTML}{E8F2FB}
\definecolor{muted}{HTML}{7FA3C4}
\definecolor{stage}{HTML}{9DC4E0}
\definecolor{bus}{HTML}{6EA8D4}
\definecolor{ctrl}{HTML}{D9A066}
\definecolor{grid}{HTML}{102031}
\definecolor{rule}{HTML}{1E3450}

\tikzset{
  blk/.style={
    draw=boxedge, fill=boxfill, rounded corners=3pt, line width=0.6pt,
    text=ink, align=left, inner sep=3.5pt,
    font=\ttfamily\scriptsize\linespread{0.92}\selectfont,
  },
  sig/.style={font=\ttfamily\tiny, text=bus, inner sep=1.2pt},
  sigc/.style={font=\ttfamily\tiny, text=ctrl, inner sep=1.2pt},
  data/.style={draw=bus, line width=0.65pt, -{Latex[length=1.5mm,width=1.1mm]}},
  ctl/.style={draw=ctrl, line width=0.65pt, dashed, dash pattern=on 2.0mm off 1.3mm,
              -{Latex[length=1.5mm,width=1.1mm]}},
  stagehead/.style={font=\ttfamily\scriptsize, text=stage, inner sep=0pt},
  colrule/.style={draw=rule, line width=0.5pt, dash pattern=on 1.2mm off 1.8mm},
}
